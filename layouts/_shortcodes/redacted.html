{{- /* Redaction shortcode. Original text is not in the page/feeds — only in Git history. Parameters (no original text): id, len, rule, geometry (inline|line|block), lines?, mode (preserve|normalized) CSS draws the surface; accessible replacement text ships for feeds/a11y. preserve ≈ fragment width; normalized = fixed sizes that hide length. */ -}} {{- $id := .Get "id" -}} {{- $rule := .Get "rule" -}} {{- $geometry := default "inline" (.Get "geometry") -}} {{- $mode := default "preserve" (.Get "mode") -}} {{- if or (not $id) (not $rule) -}} {{- errorf "%s: shortcode redacted requires id and rule" .Position -}} {{- end -}} {{- if not (in (slice "inline" "line" "block") $geometry) -}} {{- errorf "%s: shortcode redacted has unknown geometry %q (inline, line, block)" .Position $geometry -}} {{- end -}} {{- if not (in (slice "preserve" "normalized") $mode) -}} {{- errorf "%s: shortcode redacted has unknown mode %q (preserve, normalized)" .Position $mode -}} {{- end -}} {{- if eq $geometry "block" -}} {{- $lines := split (default "1,0.8,0.9" (.Get "lines")) "," -}} {{- range $lines -}} {{- $share := float (strings.TrimSpace .) -}} {{- end -}} [вымаран фрагмент текста] {{- else -}} {{- $len := int (.Get "len") -}} {{- if and (eq $mode "preserve") (lt $len 1) -}} {{- errorf "%s: shortcode redacted needs len >= 1, got %q" .Position (.Get "len") -}} {{- end -}} {{- /* 0.5em per character is the measured average of this serif at its reading size; the point is the rhythm of the line, not pixel fidelity. */ -}} {{- $width := mul $len 0.5 -}} {{- if eq $mode "normalized" -}} {{- $width = cond (lt $len 12) 4.0 (cond (lt $len 30) 9.0 16.0) -}} {{- end -}} {{- $segment := 4.0 -}} {{- $count := int (math.Ceil (div $width $segment)) -}} {{- $label := cond (eq $geometry "line") "[вымарана строка]" "[вымарано]" -}} {{- range $i := seq $count -}} {{- $last := eq $i $count -}} {{- $w := cond $last (sub $width (mul (sub $count 1) $segment)) $segment -}} {{- if not $last }}{{ end -}} {{- end -}} {{ $label }} {{- end -}}
