{{- /* Video shortcode: poster (may dither), native controls, caption. No preload/autoplay. Files under assets/. Parameters: src, poster, caption, captions? */ -}} {{- $src := .Get "src" -}} {{- $poster := .Get "poster" -}} {{- $caption := .Get "caption" -}} {{- $captions := .Get "captions" -}} {{- if not $src -}} {{- errorf "%s: video shortcode needs a src" .Page.File.Path -}} {{- end -}} {{- if not $poster -}} {{- errorf "%s: video %q needs a poster" .Page.File.Path $src -}} {{- end -}} {{- if not $caption -}} {{- errorf "%s: video %q needs a caption, which is also its accessible name" .Page.File.Path $src -}} {{- end -}} {{- $video := resources.Get (strings.TrimPrefix "/" $src) -}} {{- if not $video -}} {{- errorf "%s: video %q has no file at assets%s" .Page.File.Path $src $src -}} {{- end -}} {{- $image := partial "image-resource.html" (dict "path" $poster "context" .Page.File.Path) -}} {{- /* The rung nearest the prose column, so the poster is neither a thumbnail stretched to width nor a full-size photograph fetched to be shrunk. */ -}} {{- $width := 800 -}} {{- $entry := index hugo.Data.dither $poster -}} {{- $still := "" -}} {{- $height := 0 -}} {{- if and $entry (eq $entry.kind "photo") -}} {{- range $entry.variants -}} {{- if eq (int .width) $width -}} {{- $still = (resources.Get .path).RelPermalink -}} {{- $height = int .height -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $still -}} {{- $scaled := $image.Process (printf "resize %dx" $width) -}} {{- $still = $scaled.RelPermalink -}} {{- $height = $scaled.Height -}} {{- end -}}
