{{- /* Returns a BreadcrumbList for any page below the home page: the trail comes from .Ancestors, so it follows the real content tree instead of a hand-written path. Every "item" is an absolute URL in the site's canonical form (T56). */ -}} {{- /* A Library view lives at /library/ but its source sits in a folder that is never rendered: that folder is not a step of the trail, and the library itself is. Ancestors without an address are dropped for the same reason — a crumb that points nowhere is not a crumb. */ -}} {{- $clean := slice -}} {{- range .Ancestors.Reverse -}} {{- if .Permalink }}{{ $clean = $clean | append . }}{{ end -}} {{- end -}} {{- if .Params.view -}} {{- with site.GetPage "/library" }}{{ $clean = $clean | append . }}{{ end -}} {{- end -}} {{- $trail := $clean | append . -}} {{- $items := slice -}} {{- range $i, $p := $trail -}} {{- $items = $items | append (dict "@type" "ListItem" "position" (add $i 1) "name" (cond $p.IsHome $p.Site.Title $p.Title) "item" (partial "canonical-url.html" $p.Permalink) ) -}} {{- end -}} {{- return dict "@context" "https://schema.org" "@type" "BreadcrumbList" "itemListElement" $items -}}
