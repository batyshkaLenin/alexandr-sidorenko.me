{{- /* Preloads the font files every page paints in its first screen, so they start downloading with site.css instead of after it. Only files that are actually used belong here: a preloaded file that the page never renders is wasted. Fira Code 400 latin + cyrillic and 700 cyrillic are requested by every page. Fira Code 700 latin is not (a few publications never render it), so it stays out. PT Serif 400 latin + cyrillic set the prose of a publication, and nothing else uses them. */ -}} {{- $files := slice "fira-code-400-latin.woff2" "fira-code-400-cyrillic.woff2" "fira-code-700-cyrillic.woff2" -}} {{- if and (eq .Kind "page") (eq .Section "library") -}} {{- $files = $files | append "pt-serif-400-latin.woff2" "pt-serif-400-cyrillic.woff2" -}} {{- end -}} {{- range $files -}} {{- with resources.Get (printf "fonts/%s" .) -}} {{- else -}} {{- errorf "font-preloads: assets/fonts/%s does not exist" . -}} {{- end -}} {{- end -}}
