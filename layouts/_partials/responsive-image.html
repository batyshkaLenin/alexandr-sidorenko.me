{{- /* Responsive : WebP srcset + one JPEG fallback. Widths capped at source. Prefer WebP over AVIF here (build time / size). width/height from the fallback are mandatory for CLS. Optional square crop for avatars. */ -}} {{- $resource := .resource -}} {{- $square := .square | default false -}} {{- $limit := cond $square (math.Min $resource.Width $resource.Height) $resource.Width -}} {{- $widths := slice -}} {{- range .widths -}} {{- if le . $limit -}} {{- $widths = $widths | append . -}} {{- end -}} {{- end -}} {{- if not $widths -}} {{- $widths = slice (int $limit) -}} {{- end -}} {{- $spec := "" -}} {{- $webp := slice -}} {{- range $widths -}} {{- $spec = cond $square (printf "fill %dx%d webp" . .) (printf "resize %dx webp" .) -}} {{- $webp = $webp | append (printf "%s %dw" ($resource.Process $spec).RelPermalink .) -}} {{- end -}} {{- /* The fallback is the narrowest candidate: a browser that cannot read is also one on a small, slow device. */ -}} {{- $narrowest := index $widths 0 -}} {{- $fallback := $resource.Process (cond $square (printf "fill %dx%d" $narrowest $narrowest) (printf "resize %dx" $narrowest)) -}} {{- $class := .class | default "" -}}
