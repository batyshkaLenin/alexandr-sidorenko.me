{{- /* Resolve a site-absolute image path to its assets/ pipeline resource. Missing file → build error. Audio stays under static/ (not here). Call: {{ partial "image-resource.html" (dict "path" "/avatar.jpg" "context" .) }} */ -}} {{- $path := .path -}} {{- $context := .context -}} {{- if not (hasPrefix $path "/") -}} {{- errorf "%s: image %q must be a site-absolute path" $context $path -}} {{- end -}} {{- $resource := resources.Get (strings.TrimPrefix "/" $path) -}} {{- if not $resource -}} {{- errorf "%s: image %q has no file at assets%s" $context $path $path -}} {{- end -}} {{- return $resource -}}
