{{- /* Feed body: make root-relative href/src/srcset absolute for feed readers. Second rule rewrites `, /assets/` candidates inside srcset. */ -}} {{- $baseURL := strings.TrimSuffix "/" .Site.BaseURL -}} {{- $content := .Content -}} {{- $content = replaceRE `(href|src|srcset)="/([^/])` (printf `${1}="%s/${2}` $baseURL) $content -}} {{- $content = replaceRE `, /assets/` (printf `, %s/assets/` $baseURL) $content -}} {{- $content | safeHTML -}}
