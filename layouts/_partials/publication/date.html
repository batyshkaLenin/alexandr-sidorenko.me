{{- /* Date forms: long | short | iso | year. Always set datetime on . */ -}} {{- $date := .date -}} {{- $form := default "long" .form -}} {{- $class := default "" .class -}} {{- if not $date -}} {{- errorf "publication/date.html called without a date" -}} {{- end -}} {{- $visible := "" -}} {{- if eq $form "long" -}} {{- $visible = printf "%d %s %d" $date.Day (index hugo.Data.months (printf "%d" (int $date.Month))) $date.Year -}} {{- else if eq $form "short" -}} {{- $visible = printf "%d %s" $date.Day (index hugo.Data.months_short (printf "%d" (int $date.Month))) -}} {{- else if eq $form "iso" -}} {{- $visible = $date.Format "2006-01-02" -}} {{- else if eq $form "year" -}} {{- $visible = printf "%d" $date.Year -}} {{- else -}} {{- errorf "publication/date.html: unknown form %q (long, short, iso, year)" $form -}} {{- end -}} {{ $visible }}
