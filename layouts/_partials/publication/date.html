{{- /* One date, in the form its context asks for (§18.4). Call with a dict: {{ partial "publication/date.html" (dict "date" .Date "form" "long") }} long 15 августа 2026 inside a publication, where the date reads as text short 15 авг timeline rows and phone lines, year in the group iso 2026-08-15 dense views: index, table, machine-like lists year 2026 when the year is the only precision there is `ДД.ММ.ГГГГ` has no form here on purpose: it looks local but sorts wrong and reads as the American order to half the world (§18.4). The machine value always ships in `datetime`, whatever the visible text — so one form can change without touching what a parser sees. */ -}} {{- $date := .date -}} {{- $form := default "long" .form -}} {{- $class := default "" .class -}} {{- if not $date -}} {{- errorf "publication/date.html called without a date" -}} {{- end -}} {{- $visible := "" -}} {{- if eq $form "long" -}} {{- $visible = printf "%d %s %d" $date.Day (index hugo.Data.months (printf "%d" (int $date.Month))) $date.Year -}} {{- else if eq $form "short" -}} {{- $visible = printf "%d %s" $date.Day (index hugo.Data.months_short (printf "%d" (int $date.Month))) -}} {{- else if eq $form "iso" -}} {{- $visible = $date.Format "2006-01-02" -}} {{- else if eq $form "year" -}} {{- $visible = printf "%d" $date.Year -}} {{- else -}} {{- errorf "publication/date.html: unknown form %q (long, short, iso, year)" $form -}} {{- end -}} {{ $visible }}
