{{- /* Moves between neighbours inside one section (T55). Neighbours come from the section's own list order (.Pages.ByDate.Reverse, the same collection section.html renders) rather than .PrevInSection/.NextInSection: those follow Hugo's default page order, which puts two entries sharing a date in a different sequence than the visible list, and "следующая" would then lead somewhere the reader did not expect. A missing side renders nothing at all — a dead or empty control is worse than no control. */ -}} {{- $current := . -}} {{- with .Parent -}} {{- $pages := .Pages.ByDate.Reverse -}} {{- $index := -1 -}} {{- range $i, $page := $pages -}} {{- if eq $page.RelPermalink $current.RelPermalink -}}{{- $index = $i -}}{{- end -}} {{- end -}} {{- $prev := false -}} {{- $next := false -}} {{- if gt $index 0 -}}{{- $prev = index $pages (sub $index 1) -}}{{- end -}} {{- if and (ge $index 0) (lt (add $index 1) (len $pages)) -}}{{- $next = index $pages (add $index 1) -}}{{- end -}} {{- if or $prev $next }}
{{- with $prev }} ← предыдущая {{ .Title }} {{- end }} {{- with $next }} следующая → {{ .Title }} {{- end }}
{{- end -}} {{- end -}}
